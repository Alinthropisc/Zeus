/// Opaque handle representing a network connection owned by the engine.
/// Using an opaque integer (Handle pattern) instead of exposing raw
/// sockets/Boost types keeps this header free of transport dependencies.
using SocketHandle = std::size_t;
inline constexpr SocketHandle kInvalidSocketHandle = static_cast<SocketHandle>(-1);

/// Selects the transport strategy used by INetworkConnector::connect().
/// (Discriminator for the Strategy pattern implemented in zeus-engine.)
enum class ConnectionType : std::uint8_t { Tcp, Udp, Ssl };

/// Five-byte sentinel used for inter-process "child finished" signalling.
/// Replacement for the legacy `HYDRA_EXIT` byte array.
inline constexpr std::array<std::byte, 5> kZeusExit{
    std::byte{0xFF}, std::byte{0xFF}, std::byte{0xFF}, std::byte{0xFF}, std::byte{0xFF}};

/// A single proxy definition. Replaces the fixed-size
/// `proxy_string_ip[MAX_PROXY_COUNT]` style C arrays with a value type
/// that can be stored in a dynamically-sized container.
struct ProxyEntry {
    std::string ip;
    std::uint16_t port{0};
    std::string type;                         ///< e.g. "socks5", "http"
    std::optional<std::string> authentication; ///< "user:pass", if any
};

// ---------------------------------------------------------------------
// ExecutionContext
// ---------------------------------------------------------------------
/// Thread-safe replacement for the legacy global mutable state
/// (`debug`, `verbose`, `waittime`, `port`, `found`, `cmdlinetarget`,
/// `quiet`). Instances are injected into collaborators instead of being
/// read from process-wide globals (testability, no hidden coupling).
class ExecutionContext final {
public:
    void setDebug(bool value) noexcept { debug_.store(value, std::memory_order_relaxed); }
    [[nodiscard]] bool debug() const noexcept { return debug_.load(std::memory_order_relaxed); }

    void setVerbose(bool value) noexcept { verbose_.store(value, std::memory_order_relaxed); }
    [[nodiscard]] bool verbose() const noexcept { return verbose_.load(std::memory_order_relaxed); }

    void setQuiet(bool value) noexcept { quiet_.store(value, std::memory_order_relaxed); }
    [[nodiscard]] bool quiet() const noexcept { return quiet_.load(std::memory_order_relaxed); }

    void setWaitTime(std::chrono::seconds value) noexcept { waitTime_ = value; }
    [[nodiscard]] std::chrono::seconds waitTime() const noexcept { return waitTime_; }

    void setPort(std::uint16_t value) noexcept { port_.store(value, std::memory_order_relaxed); }
    [[nodiscard]] std::uint16_t port() const noexcept { return port_.load(std::memory_order_relaxed); }

    void incrementFound() noexcept { found_.fetch_add(1, std::memory_order_relaxed); }
    [[nodiscard]] std::int32_t found() const noexcept { return found_.load(std::memory_order_relaxed); }

    void setCommandLineTarget(std::string value) {
        std::scoped_lock lock(targetMutex_);
        cmdlineTarget_ = std::move(value);
    }
    [[nodiscard]] std::string commandLineTarget() const {
        std::scoped_lock lock(targetMutex_);
        return cmdlineTarget_;
    }

private:
    std::atomic<bool> debug_{false};
    std::atomic<bool> verbose_{false};
    std::atomic<bool> quiet_{false};
    std::chrono::seconds waitTime_{0};
    std::atomic<std::uint16_t> port_{0};
    std::atomic<std::int32_t> found_{0};
    mutable std::mutex targetMutex_;
    std::string cmdlineTarget_;
};

/// Contract for all network I/O (replacement for hydra_connect_tcp/udp/ssl,
/// hydra_send, hydra_recv*, hydra_disconnect, hydra_data_ready,
/// hydra_receive_line, hydra_register_socket).
class INetworkConnector {
public:
    virtual ~INetworkConnector() = default;

    /// Establishes a connection using the given strategy.
    /// @param sniHostname optional TLS SNI hostname (used when type == Ssl).
    [[nodiscard]] virtual std::expected<SocketHandle, std::error_code>
    connect(std::string_view host, std::uint16_t port, ConnectionType type,
            std::optional<std::string_view> sniHostname = std::nullopt) = 0;

    /// Upgrades an already-established plain TCP handle to TLS (STARTTLS).
    [[nodiscard]] virtual std::expected<void, std::error_code>
    upgradeToSsl(SocketHandle handle, std::string_view sniHostname) = 0;

    virtual void disconnect(SocketHandle handle) = 0;

    [[nodiscard]] virtual bool
    isDataReady(SocketHandle handle, std::chrono::milliseconds timeout) const = 0;

    [[nodiscard]] virtual std::expected<std::size_t, std::error_code>
    send(SocketHandle handle, std::span<const std::byte> data) = 0;

    [[nodiscard]] virtual std::expected<std::size_t, std::error_code>
    receive(SocketHandle handle, std::span<std::byte> buffer, bool blocking) = 0;

    [[nodiscard]] virtual std::expected<std::string, std::error_code>
    receiveLine(SocketHandle handle) = 0;

    /// Registers a handle for external cleanup tracking (e.g. on SIGTERM).
    virtual void registerSocket(SocketHandle handle) = 0;
};

/// Internal per-connection state: one of TCP socket / UDP socket / TLS
/// stream, plus a read buffer used by receiveLine().
struct NetworkConnector::Connection {
    using SslStream = boost::asio::ssl::stream<boost::asio::ip::tcp::socket>;
    std::variant<boost::asio::ip::tcp::socket, boost::asio::ip::udp::socket, SslStream> transport;
    boost::asio::streambuf readBuffer;

    explicit Connection(boost::asio::ip::tcp::socket socket) : transport(std::move(socket)) {}
    explicit Connection(boost::asio::ip::udp::socket socket) : transport(std::move(socket)) {}
    explicit Connection(SslStream stream) : transport(std::move(stream)) {}
};

NetworkConnector::NetworkConnector(boost::asio::io_context& ioContext)
    : ioContext_(ioContext), sslContext_(boost::asio::ssl::context::tls_client) {
    sslContext_.set_default_verify_paths();
}

NetworkConnector::~NetworkConnector() = default;

std::expected<SocketHandle, std::error_code> NetworkConnector::connect(
    std::string_view host, std::uint16_t port, ConnectionType type,
    std::optional<std::string_view> sniHostname) {
    namespace asio = boost::asio;
    using tcp = asio::ip::tcp;
    using udp = asio::ip::udp;

    boost::system::error_code ec;
    std::unique_ptr<Connection> connection;

    try {
        switch (type) {
            case ConnectionType::Tcp: {
                tcp::socket socket(ioContext_);
                tcp::resolver resolver(ioContext_);
                auto endpoints = resolver.resolve(std::string(host), std::to_string(port), ec);
                if (ec) return std::unexpected(ec);
                asio::connect(socket, endpoints, ec);
                if (ec) return std::unexpected(ec);
                connection = std::make_unique<Connection>(std::move(socket));
                break;
            }
            case ConnectionType::Udp: {
                udp::socket socket(ioContext_);
                udp::resolver resolver(ioContext_);
                auto endpoints = resolver.resolve(std::string(host), std::to_string(port), ec);
                if (ec) return std::unexpected(ec);
                socket.open(endpoints->endpoint().protocol(), ec);
                if (!ec) socket.connect(*endpoints, ec);
                if (ec) return std::unexpected(ec);
                connection = std::make_unique<Connection>(std::move(socket));
                break;
            }
            case ConnectionType::Ssl: {
                tcp::socket rawSocket(ioContext_);
                tcp::resolver resolver(ioContext_);
                auto endpoints = resolver.resolve(std::string(host), std::to_string(port), ec);
                if (ec) return std::unexpected(ec);
                asio::connect(rawSocket, endpoints, ec);
                if (ec) return std::unexpected(ec);

                Connection::SslStream sslStream(std::move(rawSocket), sslContext_);
                if (sniHostname) {
                    ::SSL_set_tlsext_host_name(sslStream.native_handle(), std::string(*sniHostname).c_str());
                }
                sslStream.handshake(asio::ssl::stream_base::client, ec);
                if (ec) return std::unexpected(ec);
                connection = std::make_unique<Connection>(std::move(sslStream));
                break;
            }
        }
    } catch (const std::exception&) {
        return std::unexpected(std::make_error_code(std::errc::connection_aborted));
    }

    std::unique_lock lock(mutex_);
    const auto handle = nextHandle_.fetch_add(1, std::memory_order_relaxed);
    connections_.emplace(handle, std::move(connection));
    return handle;
}

std::expected<void, std::error_code> NetworkConnector::upgradeToSsl(
    SocketHandle handle, std::string_view sniHostname) {
    std::unique_lock lock(mutex_);
    auto it = connections_.find(handle);
    if (it == connections_.end())
        return std::unexpected(std::make_error_code(std::errc::bad_file_descriptor));

    auto* tcpSocket = std::get_if<boost::asio::ip::tcp::socket>(&it->second->transport);
    if (!tcpSocket)
        return std::unexpected(std::make_error_code(std::errc::not_supported));

    Connection::SslStream sslStream(std::move(*tcpSocket), sslContext_);
    ::SSL_set_tlsext_host_name(sslStream.native_handle(), std::string(sniHostname).c_str());

    boost::system::error_code ec;
    sslStream.handshake(boost::asio::ssl::stream_base::client, ec);
    if (ec) return std::unexpected(ec);

    it->second->transport = std::move(sslStream);
    return {};
}

void NetworkConnector::disconnect(SocketHandle handle) {
    std::unique_lock lock(mutex_);
    connections_.erase(handle);
}

NetworkConnector::Connection* NetworkConnector::find(SocketHandle handle) const {
    std::shared_lock lock(mutex_);
    auto it = connections_.find(handle);
    return it == connections_.end() ? nullptr : it->second.get();
}

bool NetworkConnector::isDataReady(SocketHandle handle, std::chrono::milliseconds) const {
    auto* connection = find(handle);
    if (!connection) return false;

    return std::visit(
        [](auto& transport) -> bool {
            using T = std::decay_t<decltype(transport)>;
            if constexpr (std::is_same_v<T, boost::asio::ip::tcp::socket> ||
                          std::is_same_v<T, boost::asio::ip::udp::socket>) {
                return transport.available() > 0;
            } else {
                return transport.next_layer().available() > 0;
            }
        },
        connection->transport);
}

std::expected<std::size_t, std::error_code> NetworkConnector::send(
    SocketHandle handle, std::span<const std::byte> data) {
    auto* connection = find(handle);
    if (!connection) return std::unexpected(std::make_error_code(std::errc::bad_file_descriptor));

    boost::system::error_code ec;
    std::size_t written = 0;
    std::visit(
        [&](auto& transport) {
            written = boost::asio::write(transport, boost::asio::buffer(data.data(), data.size()), ec);
        },
        connection->transport);

    if (ec) return std::unexpected(ec);
    return written;
}

std::expected<std::size_t, std::error_code> NetworkConnector::receive(
    SocketHandle handle, std::span<std::byte> buffer, bool blocking) {
    auto* connection = find(handle);
    if (!connection) return std::unexpected(std::make_error_code(std::errc::bad_file_descriptor));

    boost::system::error_code ec;
    std::size_t bytesRead = 0;
    std::visit(
        [&](auto& transport) {
            using T = std::decay_t<decltype(transport)>;
            if (!blocking) {
                boost::asio::ip::tcp::socket* raw = nullptr;
                if constexpr (std::is_same_v<T, boost::asio::ip::tcp::socket>) raw = &transport;
                else if constexpr (!std::is_same_v<T, boost::asio::ip::udp::socket>) raw = &transport.next_layer();
                if (raw && raw->available() == 0) {
                    ec = boost::asio::error::would_block;
                    return;
                }
            }
            bytesRead = transport.read_some(boost::asio::buffer(buffer.data(), buffer.size()), ec);
        },
        connection->transport);

    if (ec) return std::unexpected(ec);
    return bytesRead;
}

std::expected<std::string, std::error_code> NetworkConnector::receiveLine(SocketHandle handle) {
    auto* connection = find(handle);
    if (!connection) return std::unexpected(std::make_error_code(std::errc::bad_file_descriptor));

    boost::system::error_code ec;
    std::visit(
        [&](auto& transport) {
            boost::asio::read_until(transport, connection->readBuffer, '\n', ec);
        },
        connection->transport);

    if (ec && ec != boost::asio::error::eof) return std::unexpected(ec);

    std::istream stream(&connection->readBuffer);
    std::string line;
    std::getline(stream, line);
    if (!line.empty() && line.back() == '\r') line.pop_back();
    return line;
}

CredentialProvider::CredentialProvider(std::vector<std::string> logins, std::vector<std::string> passwords)
    : logins_(std::move(logins)), passwords_(std::move(passwords)) {}

std::optional<std::pair<std::string, std::string>> CredentialProvider::nextPair() {
    std::scoped_lock lock(mutex_);
    if (loginIndex_ >= logins_.size()) return std::nullopt;

    auto login = logins_[loginIndex_];
    auto password = passwordIndex_ < passwords_.size() ? passwords_[passwordIndex_] : std::string{};

    if (++passwordIndex_ >= passwords_.size()) {
        passwordIndex_ = 0;
        ++loginIndex_;
    }
    return std::make_pair(std::move(login), std::move(password));
}

std::optional<std::string> CredentialProvider::nextLogin() {
    std::scoped_lock lock(mutex_);
    if (loginIndex_ >= logins_.size()) return std::nullopt;
    return logins_[loginIndex_++];
}

std::optional<std::string> CredentialProvider::nextPassword() {
    std::scoped_lock lock(mutex_);
    if (passwordIndex_ >= passwords_.size()) return std::nullopt;
    return passwords_[passwordIndex_++];
}

void CredentialProvider::completedPair() { attempts_.fetch_add(1, std::memory_order_relaxed); }

void CredentialProvider::completedPairFound() {
    attempts_.fetch_add(1, std::memory_order_relaxed);
    found_.fetch_add(1, std::memory_order_relaxed);
}

void CredentialProvider::completedPairSkip() { skipped_.fetch_add(1, std::memory_order_relaxed); }

// =======================================================================

std::string StringUtils::toLower(std::string_view input) const {
    std::string result(input);
    boost::algorithm::to_lower(result);
    return result;
}

std::string StringUtils::toBase64(std::span<const std::byte> data) const {
    using namespace boost::archive::iterators;
    using Base64Iterator = base64_from_binary<transform_width<const unsigned char*, 6, 8>>;

    const auto* begin = reinterpret_cast<const unsigned char*>(data.data());
    std::string encoded(Base64Iterator(begin), Base64Iterator(begin + data.size()));
    encoded.append((3 - data.size() % 3) % 3, '=');
    return encoded;
}

std::optional<std::size_t> StringUtils::caseInsensitiveFind(
    std::string_view haystack, std::string_view needle) const {
    if (needle.empty()) return 0;
    auto it = boost::ifind_first(haystack, needle);
    if (it.begin() == haystack.end()) return std::nullopt;
    return static_cast<std::size_t>(std::distance(haystack.begin(), it.begin()));
}

std::optional<std::size_t> StringUtils::memorySearch(
    std::span<const std::byte> haystack, std::span<const std::byte> needle) const {
    if (needle.empty() || needle.size() > haystack.size()) return std::nullopt;
    auto it = std::search(haystack.begin(), haystack.end(), needle.begin(), needle.end());
    if (it == haystack.end()) return std::nullopt;
    return static_cast<std::size_t>(std::distance(haystack.begin(), it));
}

std::string StringUtils::replace(
    std::string_view source, std::string_view target, std::string_view replacement) const {
    std::string result(source);
    if (!target.empty()) boost::replace_all(result, target, replacement);
    return result;
}

std::string StringUtils::dumpHex(std::span<const std::byte> data) const {
    static constexpr char kHexDigits[] = "0123456789ABCDEF";
    std::string result;
    result.reserve(data.size() * 2);
    for (auto byte : data) {
        auto value = std::to_integer<unsigned char>(byte);
        result.push_back(kHexDigits[value >> 4]);
        result.push_back(kHexDigits[value & 0x0F]);
    }
    return result;
}

std::string StringUtils::addressToString(std::string_view rawAddress, bool /*beautiful*/) const {
    try {
        return boost::asio::ip::make_address(std::string(rawAddress)).to_string();
    } catch (const std::exception&) {
        return std::string(rawAddress);
    }
}

bool StringUtils::matchesPattern(std::string_view input, std::string_view regex) const {
    try {
        return boost::regex_search(input.begin(), input.end(), boost::regex(std::string(regex)));
    } catch (const boost::regex_error&) {
        return false;
    }
}

std::string StringUtils::replacePattern(
    std::string_view input, std::string_view regex, std::string_view replacement) const {
    try {
        boost::regex pattern(std::string(regex));
        return boost::regex_replace(std::string(input), pattern, std::string(replacement));
    } catch (const boost::regex_error&) {
        return std::string(input);
    }
}

void ProxyManager::addProxy(ProxyEntry entry) {
    std::unique_lock lock(mutex_);
    proxies_.push_back(std::move(entry));
}

const std::vector<ProxyEntry>& ProxyManager::proxies() const noexcept { return proxies_; }

void ProxyManager::setUseProxy(bool value) noexcept { useProxy_.store(value, std::memory_order_relaxed); }
bool ProxyManager::useProxy() const noexcept { return useProxy_.load(std::memory_order_relaxed); }

void ProxyManager::selectProxy(std::size_t index) {
    std::unique_lock lock(mutex_);
    if (index < proxies_.size()) selectedIndex_ = index;
}

std::optional<ProxyEntry> ProxyManager::selectedProxy() const {
    std::shared_lock lock(mutex_);
    if (!selectedIndex_ || *selectedIndex_ >= proxies_.size()) return std::nullopt;
    return proxies_[*selectedIndex_];
}

std::unique_ptr<Engine> EngineFactory::createDefault(
    boost::asio::io_context& ioContext, std::vector<std::string> logins, std::vector<std::string> passwords) {

        .proxyManager = std::make_shared<ProxyManager>(),
        .processController = std::make_shared<ProcessController>(connector),
    };

    return std::make_unique<Engine>(std::move(deps));
}

// ---- Integration test: real loopback TCP round-trip ----

TEST(NetworkConnectorIntegrationTest, TcpLoopbackEchoWorks) {
    namespace asio = boost::asio;
    asio::io_context io;
    asio::ip::tcp::acceptor acceptor(io, asio::ip::tcp::endpoint(asio::ip::tcp::v4(), 0));
    const auto boundPort = acceptor.local_endpoint().port();

    std::thread server([&] {
        asio::ip::tcp::socket socket(io);
        acceptor.accept(socket);
        std::array<char, 64> buffer{};
        boost::system::error_code ec;
        std::size_t length = socket.read_some(asio::buffer(buffer), ec);
        if (!ec) asio::write(socket, asio::buffer(buffer, length), ec);
    });

    NetworkConnector connector(io);
    auto handle = connector.connect("127.0.0.1", boundPort, ConnectionType::Tcp, std::nullopt);
    ASSERT_TRUE(handle.has_value());

    std::string message = "ping";
    auto sent = connector.send(*handle, std::as_bytes(std::span(message)));
    ASSERT_TRUE(sent.has_value());

    std::array<std::byte, 64> readBuffer{};
    auto received = connector.receive(*handle, readBuffer, true);
    ASSERT_TRUE(received.has_value());

    std::string reply(reinterpret_cast<const char*>(readBuffer.data()), *received);
    EXPECT_EQ(reply, "ping");

    connector.disconnect(*handle);
    server.join();
}
